```powershell
# Create archive folder

    "BRANCH_CONSOLIDATION_SUMMARY.md",
    "CONSOLIDATION_SUMMARY.md",
    "DATABASE_OPTIMIZATION_FINAL.md",
    # ... (list all from above)
)

foreach ($doc in $obsoleteDocs) {
    Move-Item "docs\$doc" "docs\archive\v2.1\" -Force
}
```

### Phase 2: Script Archive
```powershell
# Create archive folder
New-Item -Path "utilities\archive\v2.1" -ItemType Directory -Force

# Move obsolete scripts
$obsoleteScripts = @(
    "AnalyzeFreshScan.ps1",
    "AnalyzeGroupIDs.ps1",
    # ... (list all from above)

foreach ($script in $obsoleteScripts) {

```

### Phase 3: Data File Cleanup
```powershell
# Create archive folder for analysis files
New-Item -Path "data\archive\v2.1" -ItemType Directory -Force

# Move analysis files
$analysisFiles = @(
    "BlankDescriptions_Analysis.json",
    # ... (list all analysis files)
)

foreach ($file in $analysisFiles) {
    Move-Item "data\$file" "data\archive\v2.1\" -Force
}

# Delete truly temporary files (old versions)
Remove-Item "data\API_to_GameID_Mapping_2025-11-02_*.json"
Remove-Item "data\AscensionVanity OLD.lua"
Remove-Item "data\*.bak"
```

### Phase 4: Code Review
```powershell
# Search for debug code
Get-ChildItem "AscensionVanity\*.lua" | Select-String "print\(" | Where-Object { $_.Line -notmatch "--|Print\(" }

# Search for TODOs
Get-ChildItem "AscensionVanity\*.lua" | Select-String "TODO|FIXME|HACK|XXX"

# Search for commented code blocks
Get-ChildItem "AscensionVanity\*.lua" | Select-String "--\[\[" -Context 0,3
```
